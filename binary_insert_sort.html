折半插入排序 (Binary Insertion Sort)

二分查找优化 | 稳定排序 | O(n²) 时间复杂度 | O(1) 空间复杂度

源码 (binary_insert_sort.c)

#include <stdio.h>

void BinaryInsertSort(int A[], int n) {
    int i, j, low, high, mid;
    for (i = 2; i <= n; i++) {
        A[0] = A[i];
        low = 1;
        high = i - 1;
        while (low <= high) {
            mid = (low + high) / 2;
            if (A[mid] > A[0])
                high = mid - 1;
            else
                low = mid + 1;
        }
        for (j = i - 1; j >= high + 1; j--)
            A[j + 1] = A[j];
        A[high + 1] = A[0];
    }
}

void PrintArray(int A[], int n) {
    for (int i = 1; i <= n; i++)
        printf("%d ", A[i]);
    printf("\n");
}

int main() {
    int A[9] = {0, 49, 38, 65, 97, 76, 13, 27, 49};
    int n = 8;

    printf("Before sorting: ");
    PrintArray(A, n);

    BinaryInsertSort(A, n);

    printf("After sorting:  ");
    PrintArray(A, n);

    return 0;
}
