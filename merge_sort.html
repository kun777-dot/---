源码 (merge_sort.c)

合并有序区间 Merge

void Merge(int A[], int temp[], int low, int mid, int high) {
    int i = low;
    int j = mid + 1;
    int k = low;

    while (i <= mid && j <= high) {
        if (A[i] <= A[j])
            temp[k++] = A[i++];
        else
            temp[k++] = A[j++];
    }
    while (i <= mid)
        temp[k++] = A[i++];
    while (j <= high)
        temp[k++] = A[j++];
    for (int p = low; p <= high; p++)
        A[p] = temp[p];
}

递归排序 MergeSort

void MergeSort(int A[], int temp[], int low, int high) {
    if (low < high) {
        int mid = (low + high) / 2;
        MergeSort(A, temp, low, mid);
        MergeSort(A, temp, mid + 1, high);
        Merge(A, temp, low, mid, high);
    }
}
